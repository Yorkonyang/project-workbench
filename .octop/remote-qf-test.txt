   796: /**
   797:  * API Client for Project Workbench
   798:  * 前端与后端 API 交互的封装层
   799:  */
   800: 
   801: // 默认使用相对路径 /api，由 vite 开发服务器代理到后端（localhost:3000），
   802: // 这样局域网内其他电脑访问时无需把后端地址硬编码成本机 IP，也不会触发 CORS。
   803: // 生产部署可通过环境变量 VITE_API_URL 覆盖为绝对地址。
   804: const API_BASE = import.meta.env.VITE_API_URL || '/api';
   805: 
   806: class ApiClient {
   807:     constructor() {
   808:         this.baseURL = API_BASE;
   809:     }
   810: 
   811:     async request(endpoint, options = {}) {
   812:         const url = `${this.baseURL}${endpoint}`;
   813:         const config = {
   814:             headers: {
   815:                 'Content-Type': 'application/json',
   816:                 // S5：自定义头作为 CSRF 纵深防御，后端对写请求校验此头
   817:                 'X-Workbench': '1',
   818:                 ...options.headers,
   819:             },
   820:             // 身份由后端 HttpOnly 会话 cookie（workbench_session）自动携带，
   821:             // 客户端不再手动发送 x-user-id 头，防止身份伪造。
   822:             credentials: 'include',
   823:             ...options,
   824:         };
   825: 
   826:         try {
   827:             const response = await fetch(url, config);
   828:             if (!response.ok) {
   829:                 // 401 表示会话失效，交由上层提示重新登录
   830:                 // 优先透传后端返回的错误消息（如排序号冲突提示）
   831:                 let message = `HTTP ${response.status}: ${response.statusText}`;
   832:                 try {
   833:                     const data = await response.json();
   834:                     if (data && data.error) message = data.error;
   835:                 } catch (e) { /* 忽略解析失败 */ }
   836:                 throw new Error(message);
   837:             }
   838:             return await response.json();
   839:         } catch (error) {
   840:             console.error('API Request Error:', error);
   841:             throw error;
   842:         }
   843:     }
